⚙️ POST-DEPLOYMENT (Add Environment Variables)

vercel env add SECRET_KEY
# Generate: python -c "import secrets; print(secrets.token_urlsafe(32))"

vercel env add FLASK_ENV
# Set to: production

vercel env add FLASK_DEBUG
# Set to: False

vercel env add MAIL_SERVER
# Set to: smtp.gmail.com

vercel env add MAIL_PORT
# Set to: 587

vercel env add MAIL_USE_TLS
# Set to: True

vercel env add MAIL_USERNAME
# Set to: your-email@gmail.com

vercel env add MAIL_PASSWORD
# Set to: your-app-password (from Gmail app passwords)

vercel env add MAIL_DEFAULT_SENDER
# Set to: noreply@campusrecoveryhub.com

THEN: vercel --prod
[Redeploys with environment variables]
